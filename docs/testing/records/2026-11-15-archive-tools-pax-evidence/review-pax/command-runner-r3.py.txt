from pathlib import Path
from datetime import datetime,timezone
import subprocess,os,signal,time,json,hashlib,sys,re
root=Path("/Users/hdh/.codex/worktrees/editor-unicode-codec-20261114/keelshell")
r=root/"work/archive-tools-pax-20261115/review-pax"
phase,label=sys.argv[1:3]
start=json.loads((r/"freeze-r3-start.json").read_text())
binary=Path(start["binaries"][label]["path"])
hash_before=hashlib.sha256(binary.read_bytes()).hexdigest()
source=root/("crates/keelshell-core/tests/archive_pax.rs" if label=="pax" else "crates/keelshell-core/tests/archive.rs")
source_names=set(re.findall(r"#\[test\]\s*fn\s+(\w+)\s*\(",source.read_text()))
if phase=="execute":
 inventory=json.loads((r/(label+"-inventory-r3.json")).read_text())
 assert inventory["sha256_before"]==hash_before and inventory["all_names_match"]
command=[str(binary),"--list" if phase=="list" else "--test-threads=1"]
log=r/(label+"-"+phase+"-r3.log")
assert not log.exists()
env={**os.environ,"TMPDIR":str(r/"tmp"),"TMP":str(r/"tmp"),"TEMP":str(r/"tmp")}
timed_out=False
begin=datetime.now(timezone.utc).isoformat()
t=time.monotonic()
with log.open("wb") as out:
 child=subprocess.Popen(command,cwd=root,env=env,stdout=out,stderr=subprocess.STDOUT,start_new_session=True)
 try: rc=child.wait(timeout=25 if phase=="list" else 50)
 except subprocess.TimeoutExpired:
  timed_out=True
  os.killpg(child.pid,signal.SIGKILL)
  rc=child.wait(timeout=5)
elapsed=time.monotonic()-t
text=log.read_text()
actual_names=({line.removesuffix(": test") for line in text.splitlines() if line.endswith(": test")} if phase=="list" else set(re.findall(r"^test (\w+) \.\.\. .*ok$",text,re.M)))
result={"label":label,"phase":phase,"cwd":str(root),"command":command,"started_utc":begin,"elapsed_seconds":elapsed,"returncode":rc,"timed_out":timed_out,"waited_and_reaped":child.returncode is not None,"log":str(log),"sha256_before":hash_before,"sha256_after":hashlib.sha256(binary.read_bytes()).hexdigest(),"binary_matches_r3_start":hash_before==start["binaries"][label]["sha256"],"source_test_count":len(source_names),"actual_test_count":len(actual_names),"missing_names":sorted(source_names-actual_names),"unexpected_names":sorted(actual_names-source_names),"all_names_match":actual_names==source_names,"names":sorted(actual_names)}
if phase=="execute":
 summary=re.search(r"test result: ok\. (\d+) passed; (\d+) failed; (\d+) ignored;",text)
 result["summary"]={"passed":int(summary[1]),"failed":int(summary[2]),"ignored":int(summary[3])} if summary else None
if phase=="list":
 (r/(label+"-inventory-r3.json")).write_text(json.dumps(result,ensure_ascii=False,indent=2)+"\n")
with (r/"commands-r3.jsonl").open("a") as out: out.write(json.dumps(result,ensure_ascii=False)+"\n")
print(json.dumps({k:v for k,v in result.items() if k!="names"},ensure_ascii=False,indent=2))
if phase=="execute": print(text)
assert rc==0 and not timed_out and result["all_names_match"] and result["sha256_after"]==hash_before and len(source_names)==40
if phase=="execute": assert result["summary"]=={"passed":40,"failed":0,"ignored":0}
