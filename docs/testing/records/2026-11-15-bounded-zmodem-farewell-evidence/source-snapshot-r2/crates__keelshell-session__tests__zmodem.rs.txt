fn header(kind: FrameType, position: u32) -> Vec<u8> {
    encode_header(Header::new(kind, position, HeaderFormat::Hex))
}

fn offer(name: &str, size: u64) -> TestResult<Vec<u8>> {
    let mut bytes = header(FrameType::RequestInit, 0);
    bytes.extend(encode_header(Header::new(
        FrameType::File,
        0,
        HeaderFormat::Binary32,
    )));
    bytes.extend(encode_data(
        format!("{name}\0{size} 0 100600 0 1 {size}\0").as_bytes(),
        FrameEnd::EndAck,
        HeaderFormat::Binary32,
    )?);
    Ok(bytes)
}

fn contents(bytes: &[u8]) -> TestResult<Vec<u8>> {
    let mut wire = encode_header(Header::new(FrameType::Data, 0, HeaderFormat::Binary32));
    if bytes.len() > 8192 {
        return Err("wire fixture uses one bounded subpacket".into());
    }
    wire.extend(encode_data(
        bytes,
        FrameEnd::EndAck,
        HeaderFormat::Binary32,
    )?);
    wire.extend(header(FrameType::EndOfFile, bytes.len() as u32));
    Ok(wire)
}

async fn waiting_for_farewell(fixture: &Fixture) -> TestResult<(Controller, Uuid)> {
    let mut controller = Controller::new(fixture.session.clone(), "farewell fixture");
    controller
        .feed(&fixture.session, &header(FrameType::RequestInit, 0))
        .await;
    let id = controller.lease_id().ok_or("missing receive lease")?;
    let output = controller
        .feed(&fixture.session, &header(FrameType::Finish, 0))
        .await;
    assert!(!output.writes.is_empty());
    assert!(output.events.is_empty());
    Ok((controller, id))
}

fn confirm_protocol_output(controller: &mut Controller) -> TestResult {
    let token = controller
        .protocol_output_token()
        .ok_or("missing protocol output token")?;
    controller.protocol_output_flushed(token);
    Ok(())
}

#[tokio::test]
async fn zmodem_missing_farewell_clean_exit_requires_the_exact_flushed_lease() -> TestResult {
    let fixture = Fixture::start().await?;
    let (unrelated, _) = waiting_for_farewell(&fixture).await?;
    let foreign_token = unrelated
        .protocol_output_token()
        .ok_or("missing foreign output token")?;
    for acknowledged in [false, true] {
        let (mut controller, id) = waiting_for_farewell(&fixture).await?;
        controller.protocol_output_flushed(foreign_token);
        if acknowledged {
            confirm_protocol_output(&mut controller)?;
        }
        let output = controller.transport_exited(&fixture.session, 0, true);
        assert_eq!(
            output.events.iter().any(
                |event| matches!(event, Ui::EndedWithoutFarewell { id: found } if *found == id)
            ),
            acknowledged,
        );
        assert!(
            !output
                .events
                .iter()
                .any(|event| matches!(event, Ui::Finished { .. }))
        );
        assert_eq!(
            controller.state(),
            if acknowledged {
                State::EndedWithoutFarewell
            } else {
                State::Failed
            },
        );
        assert!(!controller.is_active());
        assert!(
            controller
                .transport_exited(&fixture.session, 0, true)
                .events
                .is_empty()
        );
    }
    fixture.assert_no_authority_used();
    fixture.finish().await
}

#[cfg(any(target_os = "linux", target_os = "macos"))]
#[tokio::test]
async fn zmodem_missing_farewell_grace_preserves_saved_bytes_prompt_and_older_unknown() -> TestResult
{
    let fixture = Fixture::start().await?;
    let (_temp, root) = root()?;
    let destination = root.join("verified-empty.bin");
    let mut controller = Controller::new(fixture.session.clone(), "saved farewell fixture");
    let (id, _, review) =
        prepare_receive(&fixture, &mut controller, "empty.bin", 0, &destination).await?;
    controller
        .control(
            &fixture.session,
            Control::Confirm {
                transfer: id,
                review,
            },
        )
        .await;
    let saved = controller
        .feed(&fixture.session, &header(FrameType::EndOfFile, 0))
        .await;
    assert!(
        saved
            .events
            .iter()
            .any(|event| matches!(event, Ui::FileSaved { bytes: 0, .. }))
    );
    assert_eq!(std::fs::read(&destination)?, b"");
    let output = controller
        .feed(&fixture.session, &header(FrameType::Finish, 0))
        .await;
    assert!(!output.writes.is_empty());
    let old_token = controller
        .protocol_output_token()
        .ok_or("missing first FIN output token")?;
    controller.protocol_output_flushed(old_token);
    assert!(
        controller.tick().events.is_empty(),
        "the grace period must actually elapse"
    );
    let prompt = b"\x1b[32moperator@host\x1b[0m $ ";
    for byte in prompt {
        let output = controller.feed(&fixture.session, &[*byte]).await;
        assert!(output.events.is_empty());
        assert!(output.terminal.is_empty());
    }
    // An additional valid FIN queues another real reply; an earlier flush
    // acknowledgement must not authorize ending while that reply is pending.
    let output = controller
        .feed(&fixture.session, &header(FrameType::Finish, 0))
        .await;
    assert!(!output.writes.is_empty());
    tokio::time::sleep(std::time::Duration::from_millis(3100)).await;
    assert!(controller.tick().events.is_empty());
    assert!(controller.is_active());
    controller.mutation_pending().store(true, Ordering::Release);
    controller.protocol_output_flushed(old_token);
    assert!(controller.tick().events.is_empty());
    confirm_protocol_output(&mut controller)?;
    let output = controller.tick();
    assert_eq!(output.events, vec![Ui::EndedWithoutFarewell { id }]);
    assert!(
        output
            .terminal
            .windows(prompt.len())
            .any(|bytes| bytes == prompt)
    );
    assert!(!output.terminal.windows(3).any(|bytes| bytes == b"**\x18"));
    assert!(controller.mutation_pending().load(Ordering::Acquire));
    assert!(!controller.publication_running().load(Ordering::Acquire));
    assert!(!controller.is_active());
    assert_eq!(std::fs::read(&destination)?, b"");
    let output = controller.feed(&fixture.session, b"next output\n").await;
    assert_eq!(output.terminal, b"next output\n");
    assert!(output.events.is_empty());
    fixture.assert_no_authority_used();
    fixture.finish().await
}

#[tokio::test]
async fn zmodem_missing_farewell_rejects_same_lease_receipts_after_new_output() -> TestResult {
    let fixture = Fixture::start().await?;
    for stale_receipt in [true, false] {
        let (mut controller, id) = waiting_for_farewell(&fixture).await?;
        let old = controller
            .protocol_output_token()
            .ok_or("missing initial reply")?;
        let output = controller
            .feed(&fixture.session, &header(FrameType::Finish, 0))
            .await;
        assert!(!output.writes.is_empty());
        let current = controller
            .protocol_output_token()
            .ok_or("missing current reply")?;
        assert_ne!(old, current);
        controller.protocol_output_flushed(if stale_receipt { old } else { current });
        let output = controller.transport_exited(&fixture.session, 0, true);
        assert_eq!(
            output.events.contains(&Ui::EndedWithoutFarewell { id }),
            !stale_receipt,
        );
        assert!(
            !output
                .events
                .iter()
                .any(|event| matches!(event, Ui::Finished { .. }))
        );
        // A late receipt after terminal failure cannot reactivate its lease.
        controller.protocol_output_flushed(current);
        assert!(!controller.is_active());
    }
    fixture.assert_no_authority_used();
    fixture.finish().await
}

fn upload_request(output: &Output) -> TestResult<Uuid> {
    output
        .events
        .iter()
        .find_map(|event| {
            if let Ui::UploadRequested { id } = event {
                Some(*id)
            } else {
                None
            }
        })
        .ok_or_else(|| format!("missing upload request: {:?}", output.events).into())
}

fn download_offer(output: &Output) -> TestResult<(Uuid, Uuid)> {
    output
        .events
        .iter()
        .find_map(|event| {
            if let Ui::DownloadOffered { id, offer, .. } = event {
                Some((*id, *offer))
            } else {
                None
            }
        })
        .ok_or_else(|| format!("missing download offer: {:?}", output.events).into())
}

#[cfg(any(target_os = "linux", target_os = "macos"))]
fn review(output: &Output) -> TestResult<Uuid> {
    output
        .events
        .iter()
        .find_map(|event| {
            if let Ui::Review { review, .. } = event {
                Some(*review)
            } else {
                None
            }
        })
        .ok_or_else(|| format!("missing review: {:?}", output.events).into())
}

fn rejected(output: &Output, reason: Failure) {
    assert!(
        output.events.iter().any(
            |event| matches!(event, Ui::ActionRejected { reason: found, .. } if *found == reason)
        ),
        "{:?}",
        output.events
    );
    assert!(!output.events.iter().any(|event| matches!(
        event,
        Ui::Failed { .. } | Ui::Cancelled { .. } | Ui::Finished { .. }
    )));
    assert!(output.writes.is_empty());
}

fn root() -> TestResult<(tempfile::TempDir, PathBuf)> {
    let temp = tempfile::tempdir()?;
    let root = temp.path().canonicalize()?;
    Ok((temp, root))
}

#[cfg(any(target_os = "linux", target_os = "macos"))]
async fn prepare_receive(
    fixture: &Fixture,
    controller: &mut Controller,
    name: &str,
    size: u64,
    destination: &Path,
) -> TestResult<(Uuid, Uuid, Uuid)> {
    let output = controller.feed(&fixture.session, &offer(name, size)?).await;
    let (transfer, offer) = download_offer(&output)?;
    let output = controller
        .control(
            &fixture.session,
            Control::PrepareReceive {
                transfer,
                offer,
                destination: destination.to_owned(),
            },
        )
        .await;
    Ok((transfer, offer, review(&output)?))
}

#[tokio::test]
async fn zmodem_controller_acquires_a_unique_lease_only_after_complete_valid_detection()
-> TestResult {
    let fixture = Fixture::start().await?;
    let mut controller = Controller::new(fixture.session.clone(), "fixture");
    let request = header(FrameType::ReceiverInit, 0);
    let mut events = Vec::new();
    for (index, byte) in request.iter().enumerate() {
        let output = controller.feed(&fixture.session, &[*byte]).await;
        events.extend(output.events);
        if index + 3 < request.len() {
            assert!(!controller.is_active());
        }
    }
    let id = upload_request(&Output {
        events,
        ..Output::default()
    })?;
    assert_ne!(id, Uuid::nil());
    assert_eq!(controller.lease_id(), Some(id));
    assert_eq!(controller.state(), State::AwaitingSources);
    let output = controller
        .control(&fixture.session, Control::Cancel { transfer: id })
        .await;
    assert!(output.events.contains(&Ui::Cancelled {
        id,
        uncertain: false
    }));
    assert_eq!(controller.state(), State::Cancelled);
    assert!(!controller.is_active());
    assert_eq!(controller.lease_id(), None);
    assert!(
        output
            .writes
            .iter()
            .any(|bytes| bytes.starts_with(&[24; 8]))
    );
    fixture.assert_no_authority_used();
    fixture.finish().await
}

#[cfg(any(target_os = "linux", target_os = "macos"))]
#[tokio::test]
async fn zmodem_controller_local_claims_exclude_other_connections_and_existing_sftp_scopes()
-> TestResult {
    let first = Fixture::start().await?;
    let second = Fixture::start().await?;
    let (_temp, root) = root()?;
    let target = root.join("shared.txt");
    let mut one = Controller::new(first.session.clone(), "first");
    let mut two = Controller::new(second.session.clone(), "second");
    let (transfer_one, _, review_one) =
        prepare_receive(&first, &mut one, "one.txt", 3, &target).await?;
    one.control(
        &first.session,
        Control::Confirm {
            transfer: transfer_one,
            review: review_one,
        },
    )
    .await;
    assert_eq!(one.state(), State::Receiving);
    let (transfer_two, offer_two, review_two) =
        prepare_receive(&second, &mut two, "two.txt", 3, &target).await?;
    let output = two
        .control(
            &second.session,
            Control::Confirm {
                transfer: transfer_two,
                review: review_two,
            },
        )
        .await;
    rejected(&output, Failure::DestinationBusy);
    assert!(two.is_active());
    let sftp = second.session.sftp().await?;
    let spec = TransferSpec::download("/", &target);
    assert!(matches!(
        sftp.reserve_directory_sync(&spec, &|| true).await,
        Err(SessionError::MutationBusy)
    ));
    let output = one
        .control(
            &first.session,
            Control::Cancel {
                transfer: transfer_one,
            },
        )
        .await;
    assert!(output.events.contains(&Ui::Cancelled {
        id: transfer_one,
        uncertain: false
    }));
    let output = two
        .control(
            &second.session,
            Control::PrepareReceive {
                transfer: transfer_two,
                offer: offer_two,
                destination: target.clone(),
            },
        )
        .await;
    let fresh = review(&output)?;
    let output = two
        .control(
            &second.session,
            Control::Confirm {
                transfer: transfer_two,
                review: fresh,
            },
        )
        .await;
    assert!(!output.writes.is_empty());
    assert_eq!(two.state(), State::Receiving);
    assert!(!target.exists());
    two.control(
        &second.session,
        Control::Cancel {
            transfer: transfer_two,
        },
    )
    .await;
    assert!(sftp.reserve_directory_sync(&spec, &|| true).await.is_ok());
    sftp.close().await?;
    assert_eq!(first.observed.commands.load(Ordering::Acquire), 0);
    assert_eq!(second.observed.commands.load(Ordering::Acquire), 0);
    first.finish().await?;
    second.finish().await
}
