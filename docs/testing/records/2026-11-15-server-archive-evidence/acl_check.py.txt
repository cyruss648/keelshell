"""Run the real inherited-ACL regression without acquiring the Cargo slot."""
from pathlib import Path
import subprocess
import sys

root = Path(__file__).resolve().parents[2]
text = (root / "crates/keelshell-core/tests/server_archive.rs").read_text()
suite = text.split('const PYTHON_SUITE: &str = r####"', 1)[1].split('"####;', 1)[0]
suite = suite.replace("unittest.main(verbosity=2)", "unittest.main(defaultTest='HelperTests.test_darwin_inherited_acl_is_removed_before_source_bytes_and_publication', verbosity=2)")
helper = Path(sys.argv[1]).read_text() if len(sys.argv) > 1 else (
    root / "crates/keelshell-core/src/server_archive.py").read_text()
raise SystemExit(subprocess.run([sys.executable, "-B", "-c", suite, helper], timeout=25).returncode)
