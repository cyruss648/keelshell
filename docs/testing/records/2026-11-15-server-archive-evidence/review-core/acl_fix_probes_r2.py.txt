def execute(arguments, data=b"", timeout=12):
    started = time.monotonic()
    child = subprocess.Popen(arguments, stdin=subprocess.PIPE, stdout=subprocess.PIPE,
                             stderr=subprocess.PIPE, env=ENV, start_new_session=True)
    timed_out = False
    try:
        stdout, stderr = child.communicate(data, timeout=timeout)
    except subprocess.TimeoutExpired:
        timed_out = True
        os.killpg(child.pid, signal.SIGKILL)
        stdout, stderr = child.communicate()
    finally:
        try:
            os.killpg(child.pid, signal.SIGKILL)
        except ProcessLookupError:
            pass
        child.wait()
    COMMANDS.append({
        "command": arguments, "stdin_sha256": hashlib.sha256(data).hexdigest(),
        "exit_code": child.returncode, "timed_out": timed_out,
        "seconds": time.monotonic() - started,
        "stdout": stdout.decode("utf-8", "backslashreplace"),
        "stderr": stderr.decode("utf-8", "backslashreplace"),
        "process_group_reaped": True,
    })
    return child.returncode, stdout, stderr


def check(name, function, native):
    first_command = len(COMMANDS)
    try:
        detail = function()
        CASES.append({"case": name, "passed": True, "native_darwin": native,
                      "detail": detail,
                      "commands": list(range(first_command, len(COMMANDS)))})
        print("PASS", name, flush=True)
    except Exception as error:
        CASES.append({"case": name, "passed": False, "native_darwin": native,
                      "error": repr(error),
                      "commands": list(range(first_command, len(COMMANDS)))})
        print("FAIL", name, repr(error), flush=True)


def listing(path):
    code, stdout, stderr = execute(["/bin/ls", "-led", str(path)])
    assert code == 0 and not stderr
    return stdout.decode()


def acl_lines(path):
    return listing(path).splitlines()[1:]


def publication(format_name, fail_link=False):
    parent = FIXTURES / (format_name + ("-link-failure" if fail_link else "-published"))
    parent.mkdir()
    source = parent / "private-source"
    source.write_bytes(b"source held private before parent ACL\0\xff")
    source.chmod(0o600)
    source_before = acl_lines(source)
    assert source_before == []
    code, stdout, stderr = execute([
        "/bin/chmod", "+a",
        "everyone allow read,execute,file_inherit,directory_inherit", str(parent)
    ])
    assert code == 0 and not stdout and not stderr
    parent_before = acl_lines(parent)
    assert parent_before and "everyone" in "\n".join(parent_before)
    destination = parent / ("result." + format_name)
    stage = parent / (".keelshell-server-archive-" + str(uuid.uuid4()))
    request = {"version": 1, "sources": [str(source)],
               "destination": str(destination), "format": format_name}
    argument = base64.b64encode(json.dumps(request, ensure_ascii=False).encode()).decode()
    code, inspected, stderr = execute([sys.executable, "-B", str(HELPER),
                                      "inspect", argument])
    assert code == 0 and not stderr
    if fail_link:
        command = [sys.executable, "-B", str(REVIEW / "fault_worker_r1.py"),
                   str(HELPER), "unsupported-link", "pack", argument, str(stage)]
    else:
        command = [sys.executable, "-B", str(HELPER), "pack", argument, str(stage)]
    code, stdout, stderr = execute(command, inspected)
    if fail_link:
        assert code == 65 and stdout == b""
        assert stderr == b"KeelShell server archive rejected\n"
        assert not destination.exists() and (stage / "payload").is_file()
        assert stage.stat().st_mode & 0o7777 == 0o700
        assert (stage / "payload").stat().st_mode & 0o7777 == 0o600
        assert acl_lines(stage) == [] and acl_lines(stage / "payload") == []
    else:
        assert code == 0 and stderr == b""
        assert not stage.exists() and destination.is_file()
        assert destination.stat().st_mode & 0o7777 == 0o600
        assert acl_lines(destination) == []
        receipt = json.loads(stdout)
        encoded = destination.read_bytes()
        assert receipt["archive_bytes"] == len(encoded)
        assert receipt["sha256"] == hashlib.sha256(encoded).hexdigest()
    assert acl_lines(parent) == parent_before
    assert acl_lines(source) == source_before
    return {"fixture": str(parent), "source_acl_unchanged": True,
            "parent_acl_unchanged": True, "private_result_or_stage_acl_empty": True,
            "source_mode": "0600", "helper_returncode": code}


check("native-inherited-acl-zip-publication", lambda: publication("zip"), True)

      lambda: publication("tar.gz", True), True)


def native_errno_semantics():
    library = ctypes.CDLL(None, use_errno=True)
    for name, arguments, result in (
        ("acl_init", (ctypes.c_int,), ctypes.c_void_p),
        ("acl_free", (ctypes.c_void_p,), ctypes.c_int),
        ("acl_set_fd_np", (ctypes.c_int, ctypes.c_void_p, ctypes.c_int), ctypes.c_int),
        ("acl_get_fd_np", (ctypes.c_int, ctypes.c_int), ctypes.c_void_p),
        ("acl_valid", (ctypes.c_void_p,), ctypes.c_int),
        ("acl_get_entry", (ctypes.c_void_p, ctypes.c_int,
                           ctypes.POINTER(ctypes.c_void_p)), ctypes.c_int),
    ):
        function = getattr(library, name)
        function.argtypes, function.restype = arguments, result
    empty = library.acl_init(0)
    assert empty
    details = {}
    try:
        assert library.acl_valid(empty) == 0
        entry = ctypes.c_void_p()
        ctypes.set_errno(0)
        result = library.acl_get_entry(empty, 0, ctypes.byref(entry))
        details["acl_init_empty_first_entry"] = [result, ctypes.get_errno()]
        assert result == -1 and ctypes.get_errno() == errno.EINVAL
        ctypes.set_errno(0)
        result = library.acl_set_fd_np(-1, empty, 0x100)
        details["invalid_fd_set"] = [result, ctypes.get_errno()]
        assert result == -1 and ctypes.get_errno() == errno.EBADF
    finally:
        library.acl_free(empty)
    ctypes.set_errno(0)
    acl = library.acl_get_fd_np(-1, 0x100)
    details["invalid_fd_get"] = [bool(acl), ctypes.get_errno()]
    assert not acl and ctypes.get_errno() == errno.EBADF
    path = FIXTURES / "errno-file"
    path.write_bytes(b"reviewer-only")
    path.chmod(0o600)
    descriptor = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC)
    try:
        ctypes.set_errno(0)
        acl = library.acl_get_fd_np(descriptor, 0x100)
        details["new_valid_fd_without_acl"] = [bool(acl), ctypes.get_errno()]
        assert not acl and ctypes.get_errno() == errno.ENOENT
        MODULE["private_acl"](descriptor)
        code, _, stderr = execute(["/bin/chmod", "+a", "everyone allow read",
                                   str(path)])
        assert code == 0 and not stderr
        ctypes.set_errno(0)
        acl = library.acl_get_fd_np(descriptor, 0x100)
        assert acl
        try:
            assert library.acl_valid(acl) == 0
            entry = ctypes.c_void_p()
            ctypes.set_errno(0)
            result = library.acl_get_entry(acl, 0, ctypes.byref(entry))
            details["actual_allow_first_entry"] = [result, ctypes.get_errno()]
            assert result == 0 and entry
        finally:
            library.acl_free(acl)
        try:
            MODULE["private_acl"](descriptor)
        except MODULE["Rejected"]:
            pass
        else:
            raise AssertionError("nonempty native ACL accepted")
        MODULE["private_acl"](descriptor, clear=True)
        assert acl_lines(path) == []
        ctypes.set_errno(0)
        acl = library.acl_get_fd_np(descriptor, 0x100)
        details["valid_fd_after_empty_set"] = [bool(acl), ctypes.get_errno()]
        if acl:
            try:
                assert library.acl_valid(acl) == 0
                entry = ctypes.c_void_p()
                ctypes.set_errno(0)
                result = library.acl_get_entry(acl, 0, ctypes.byref(entry))
                details["empty_readback_first_entry"] = [result, ctypes.get_errno()]
                assert result == -1 and ctypes.get_errno() == errno.EINVAL
            finally:
                library.acl_free(acl)
        else:
            assert ctypes.get_errno() == errno.ENOENT
        for clear in (False, True):
            try:
                MODULE["private_acl"](-1, clear=clear)
            except MODULE["Rejected"]:
                pass
            else:
                raise AssertionError("invalid descriptor accepted")
    finally:
        os.close(descriptor)
    return details


check("native-empty-acl-and-error-errno", native_errno_semantics, True)

def linux_case(name, listed, remove_error=None, clear=True,
               should_reject=False, expected_removed=()):
    removed = []
    lists = iter(listed)

    def fake_list(descriptor):
        assert descriptor == 12345
        next_value = next(lists)
        if isinstance(next_value, Exception):
            raise next_value
        return next_value

    def fake_remove(descriptor, attribute):
        assert descriptor == 12345
        if remove_error:
            raise remove_error
        removed.append(attribute)

    with patch.object(sys, "platform", "linux"), \
            patch.object(os, "listxattr", fake_list, create=True), \
            patch.object(os, "removexattr", fake_remove, create=True):
        rejected = False
        try:
            MODULE["private_acl"](12345, clear=clear)
        except (MODULE["Rejected"], OSError):
            rejected = True
    assert rejected == should_reject
    assert tuple(removed) == expected_removed
    return {"emulated_branch_only": True, "native_linux": False,
            "rejected": rejected, "removed": removed, "case": name}


linux_probes = (
    ("clear-known-access-default",
     lambda: linux_case("known", [
         ["system.posix_acl_default", "system.posix_acl_access", "user.note"],
         ["user.note"]],
         expected_removed=("system.posix_acl_access", "system.posix_acl_default"))),
    ("unknown-family-rejected-before-delete",
     lambda: linux_case("unknown", [["system.nfs4_acl", "system.posix_acl_access"]],
                        should_reject=True)),
    ("non-clearing-check-rejects-known-acl",
     lambda: linux_case("read-only-check", [["system.posix_acl_access"],
                                           ["system.posix_acl_access"]],
                        clear=False, should_reject=True)),
    ("list-unavailable-rejected",
     lambda: linux_case("list-error", [OSError(errno.EOPNOTSUPP, "reviewer fault")],
                        should_reject=True)),
    ("remove-denied-rejected",
     lambda: linux_case("remove-error", [["system.posix_acl_access"]],
                        remove_error=OSError(errno.EACCES, "reviewer fault"),
                        should_reject=True)),
    ("acl-reappearing-after-clear-rejected",
     lambda: linux_case("reappear", [["system.posix_acl_access"],
                                     ["system.nfs4_acl"]],
                        should_reject=True,
                        expected_removed=("system.posix_acl_access",))),
    ("unrelated-xattrs-preserved",
     lambda: linux_case("unrelated", [["user.note", "security.selinux"],
                                      ["user.note", "security.selinux"]])),
)
for name, function in linux_probes:
    check("emulated-linux-" + name, function, False)
